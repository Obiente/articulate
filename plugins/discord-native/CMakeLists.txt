cmake_minimum_required(VERSION 3.24)
project(articulate_discord_audio LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This adapter requires Windows x64 with the MSVC ABI")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
include(FetchContent)
FetchContent_Declare(minhook
  GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
  GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(minhook)
FetchContent_Declare(node_api_headers
  GIT_REPOSITORY https://github.com/nodejs/node-api-headers.git
  GIT_TAG 8d0ef96a771d67f79b5436604d8c697ea70b618e
  SOURCE_SUBDIR unused)
FetchContent_MakeAvailable(node_api_headers)
add_library(articulate_discord_audio SHARED adapter.cpp node_addon.cpp)
set_target_properties(articulate_discord_audio PROPERTIES SUFFIX ".node")
target_include_directories(articulate_discord_audio PRIVATE "${node_api_headers_SOURCE_DIR}/include")
target_link_libraries(articulate_discord_audio PRIVATE minhook bcrypt)
target_compile_options(articulate_discord_audio PRIVATE /W4 /WX /EHsc)
target_link_options(articulate_discord_audio PRIVATE /DEBUG:NONE /INCREMENTAL:NO)
target_compile_definitions(articulate_discord_audio PRIVATE _ITERATOR_DEBUG_LEVEL=0)
add_executable(audio_adapter_test adapter.cpp adapter_test.cpp)
target_compile_definitions(audio_adapter_test PRIVATE ARTICULATE_ADAPTER_TEST _ITERATOR_DEBUG_LEVEL=0)
target_compile_options(audio_adapter_test PRIVATE /W4 /WX /EHsc)
target_link_libraries(audio_adapter_test PRIVATE minhook bcrypt)
enable_testing()
add_test(NAME audio_adapter_contract COMMAND audio_adapter_test)
find_program(NODE_EXECUTABLE node)
if(NODE_EXECUTABLE)
  add_test(NAME audio_addon_loading COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/addon_test.cjs" "$<TARGET_FILE:articulate_discord_audio>")
  add_test(NAME audio_preload_transport COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/preload_test.cjs")
endif()
install(TARGETS articulate_discord_audio RUNTIME DESTINATION .)
install(FILES "${minhook_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME MinHook.txt)
install(FILES "${node_api_headers_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME Node-API-Headers.txt)
install(FILES preload.cjs DESTINATION . RENAME articulate-audio-preload.cjs)
install(FILES "../../LICENSE" DESTINATION licenses RENAME Articulate-AGPL.txt)
